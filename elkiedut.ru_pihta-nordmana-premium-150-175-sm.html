function init() {
const root = document.getElementById('elkiedut-treecalc');
const actions = root && root.querySelector('#tc_actions');
if (!actions || root.dataset.treecalcPhotosReady || !Object.keys(photos).length) return;
root.dataset.treecalcPhotosReady = 'true';
root.classList.add('treecalc--photos');
const result = actions.parentElement;
const section = document.createElement('div');
section.className = 'treecalc__products';
const heading = document.createElement('h3');
heading.className = 'treecalc__products-title';
heading.textContent = 'Подходящие деревья';
result.after(section);
section.append(heading, actions);

function enhance() {
actions.querySelectorAll('a.treecalc__btn').forEach(function (link) {
if (link.querySelector('.treecalc__photo')) return;
let path;
try { path = new URL(link.getAttribute('href'), window.location.href).pathname; }
catch (_) { return; }
const src = photos[path];
if (typeof src !== 'string' || !src) return;

const label = document.createElement('span');
label.className = 'treecalc__card-label';
label.textContent = link.textContent;
const media = document.createElement('span');
media.className = 'treecalc__photo';
const image = document.createElement('img');
image.alt = '';
image.width = 320;
image.height = 320;
image.loading = 'lazy';
image.decoding = 'async';
image.addEventListener('error', function () {
media.remove();
link.classList.remove('treecalc__product');
}, { once: true });
image.src = src;
media.append(image);
link.replaceChildren(media, label);
link.classList.add('treecalc__product');
});
}

// The original calculator updates the links on the controls first.
// Bubbling events then add their matching photographs synchronously.
root.addEventListener('input', enhance);
root.addEventListener('change', enhance);
enhance();
}

if (document.readyState === 'loading') document.addEventListener('DOMContentLoaded', init, { once: true });
else init();
})();
